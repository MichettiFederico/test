// ========================================== // BOT DE GRILLA GEOMÉTRICA PARA SOLANA (JS) // - 180 niveles de grilla hacia abajo // - Disparador de Break-Even al 50% de llenado // - Espaciado dinámico basado en ATR // - Stop-Loss de emergencia global (-8%) // ========================================== const TOTAL_SOL = 25; // Asignación base de SOL const NIVELES = 180; // 180 órdenes en la grilla geométrica const MULTIPLICADOR = 1.015; // Progresión geométrica del tamaño const UMBRAL_BREAKEVEN = 0.50; // 50% de órdenes llenas para activar salida break-even let salidaId = 0; function init(api) { api.log("🚀 Iniciando Bot de Solana (180 niveles + Break-Even al 50%)..."); } function onCandle(c, api) { const pos = api.position; const orders = api.orders; const currentPrice = api.price; // 1. Control de Stop-Loss de Emergencia Global (-8% de caída en contra) if (pos.qty > 0 && pos.entry > 0) { const drawdown = (currentPrice - pos.entry) / pos.entry; if (drawdown <= -0.08) { api.log("🚨 [SL EMERGENCIA] Caída crítica del 8% detectada. ¡Cerrando por seguridad!"); api.cancelAll(); api.market('sell', pos.qty, { closePosition: true, tag: 'emergency_sl' }); salidaId = 0; return; } } // 2. Si no hay posición ni órdenes activas, armar la grilla inicial if (pos.qty === 0 && orders.length === 0) { armarGrilla(api, currentPrice); return; } // 3. Monitorear llenado de la grilla para activar Break-Even al 50% const buyOrdersActive = orders.filter(o => o.side === 'buy' && o.tag && o.tag.startsWith('grid_')); const filledCount = NIVELES - buyOrdersActive.length; const completionRatio = filledCount / NIVELES; if (completionRatio >= UMBRAL_BREAKEVEN && pos.qty > 0) { api.log(`🎯 [BREAK-EVEN] ¡50% de la grilla completado (${filledCount}/${NIVELES})! Intentando cerrar en precio medio: ${pos.entry.toFixed(2)}`); // Cancelar órdenes pendientes de la grilla para limpiar el tablero api.cancelAll(); // Colocar orden de salida en break-even si aún no está puesta if (!salidaId && pos.entry > 0) { salidaId = api.limit('sell', 0, pos.entry, { closePosition: true, tag: 'breakeven_tp' }); } } } function onFill(f, api) { api.log(`[FILL] Lado: ${f.side} | Qty: ${f.qty.toFixed(2)} | Precio: ${f.price.toFixed(2)} | Tag: ${f.tag}`); if (f.tag === 'breakeven_tp' || f.tag === 'emergency_sl') { salidaId = 0; api.cancelAll(); } } function armarGrilla(api, basePrice) { // Cálculo de espaciado dinámico usando ATR para adaptarse a la volatilidad de Solana const atr = api.atr(14); const stepPct = !isNaN(atr) ? Math.max((atr / basePrice) * 0.04, 0.0003) : 0.0003; // Calcular tamaño base con progresión geométrica exacta let sumPesos = 0; for (let i = 0; i < NIVELES; i++) { sumPesos += Math.pow(MULTIPLICADOR, i); } const baseSize = TOTAL_SOL / sumPesos; api.cancelAll(); salidaId = 0; // Colocar las 180 órdenes límite de compra hacia abajo for (let i = 0; i < NIVELES; i++) { const levelPrice = basePrice * (1 - (stepPct * (i + 1))); const levelSize = baseSize * Math.pow(MULTIPLICADOR, i); api.limit('buy', levelSize, levelPrice, { tag: 'grid_' + i }); } api.log(`📉 Grilla geométrica de ${NIVELES} niveles desplegada desde base ${basePrice.toFixed(2)}.`); }
