""" Backtest de 1000 estrategias aleatorias en BTCUSDT perpetuo (Binance) con 60x. - Datos: ccxt (Binance USDM). pip install ccxt pandas numpy - Optimiza en el 70% inicial (train) y valida en el 30% final (test, nunca visto). - Modela: fees taker, slippage, liquidación a 60x (conservadora) y SL siempre dentro de la distancia de liquidación. Uso: python backtest_btc_60x.py """ import json import random import ccxt import numpy as np import pandas as pd LEV = 60 FEE = 0.0005 # taker por lado, sobre nocional SLIP = 0.0002 # slippage por entrada MM = 0.005 # margen de mantenimiento aprox. MARGIN_FRAC = 0.10 # fracción del equity usada como margen por trade N_STRATS = 1000 TF = "1h" DAYS = 730 SEED = 42 MAX_SL = 1 / LEV - MM # ~1.17%: el SL debe quedar adentro de esto def fetch(tf=TF, days=DAYS): ex = ccxt.binanceusdm() since = ex.milliseconds() - days * 86_400_000 out = [] while True: b = ex.fetch_ohlcv("BTC/USDT", tf, since=since, limit=1500) if not b: break out += b since = b[-1][0] + 1 if len(b) < 2: break df = pd.DataFrame(out, columns=["ts", "open", "high", "low", "close", "vol"]) df = df.drop_duplicates("ts").reset_index(drop=True) return df def ema(s, n): return s.ewm(span=n, adjust=False).mean() def rsi(s, n): d = s.diff() up = d.clip(lower=0).ewm(alpha=1 / n, adjust=False).mean() dn = (-d.clip(upper=0)).ewm(alpha=1 / n, adjust=False).mean() return 100 - 100 / (1 + up / dn.replace(0, np.nan)) def signals(df, p): c = df["close"] if p["fam"] == "ema": f, s = ema(c, p["fast"]), ema(c, p["slow"]) long = (f > s) & (f.shift() <= s.shift()) short = (f < s) & (f.shift() >= s.shift()) elif p["fam"] == "rsi": r = rsi(c, p["n"]) long = (r > p["lo"]) & (r.shift() <= p["lo"]) short = (r < p["hi"]) & (r.shift() >= p["hi"]) else: # donchian hi = df["high"].rolling(p["n"]).max().shift() lo = df["low"].rolling(p["n"]).min().shift() long = c > hi short = c < lo return long.fillna(False).values, short.fillna(False).values def backtest(df, long, short, sl, rr, maxbars): o, h, l, c = (df[k].values for k in ("open", "high", "low", "close")) n = len(df) idx = np.flatnonzero(long | short) eq, peak, mdd = 1.0, 1.0, 0.0 trades = wins = liqs = 0 k = 0 while k < len(idx): i = idx[k] if i + 1 >= n: break side = 1 if long[i] else -1 entry = o[i + 1] * (1 + side * SLIP) sl_p = entry * (1 - side * sl) tp_p = entry * (1 + side * sl * rr) liq = entry * (1 - side * (1 / LEV - MM)) margin = eq * MARGIN_FRAC notional = margin * LEV end = min(n - 1, i + 1 + maxbars) exit_p, liquidated = c[end], False for j in range(i + 1, end + 1): lo_hit = l[j] <= liq if side == 1 else h[j] >= liq sl_hit = l[j] <= sl_p if side == 1 else h[j] >= sl_p tp_hit = h[j] >= tp_p if side == 1 else l[j] <= tp_p if lo_hit: exit_p, liquidated = liq, True break if sl_hit: # conservador: si toca SL y TP en la misma vela, gana el SL exit_p = sl_p break if tp_hit: exit_p = tp_p break if liquidated: pnl = -margin liqs += 1 else: pnl = notional * side * (exit_p / entry - 1) - notional * FEE * 2 eq += pnl trades += 1 wins += pnl > 0 peak = max(peak, eq) mdd = max(mdd, (peak - eq) / peak) if eq <= 0.01: eq = 0.0 break k = np.searchsorted(idx, j, side="right") return dict(ret=eq - 1, mdd=mdd, trades=trades, winrate=wins / trades if trades else 0, liqs=liqs) def sample_params(): fam = random.choice(["ema", "rsi", "donch"]) p = dict(fam=fam, sl=random.uniform(0.003, min(0.009, MAX_SL * 0.8)), rr=random.choice([1, 1.5, 2, 3]), maxbars=random.choice([12, 24, 48, 96])) if fam == "ema": p["fast"] = random.randint(5, 40) p["slow"] = p["fast"] + random.randint(10, 160) elif fam == "rsi": p.update(n=random.choice([7, 14, 21]), lo=random.randint(20, 40), hi=random.randint(60, 80)) else: p["n"] = random.randint(10, 120) return p def run(df, p): long, short = signals(df, p) return backtest(df, long, short, p["sl"], p["rr"], p["maxbars"]) def score(r): if r["trades"] < 30: return -9 return r["ret"] / max(r["mdd"], 0.05) if __name__ == "__main__": random.seed(SEED) df = fetch() cut = int(len(df) * 0.7) train, test = df.iloc[:cut].reset_index(drop=True), df.iloc[cut:].reset_index(drop=True) print(f"Velas: {len(df)} | train {len(train)} | test {len(test)}") res = [] for _ in range(N_STRATS): p = sample_params() res.append((p, run(train, p))) res.sort(key=lambda x: score(x[1]), reverse=True) # Evalúa TODAS en test y guarda results.json para el visor HTML (id = ranking en train) out = [] for rank, (p, rt) in enumerate(res, 1): out.append({"id": rank, "params": p, "train": rt, "test": run(test, p)}) with open("results.json", "w") as fh: json.dump({"meta": {"tf": TF, "lev": LEV, "days": DAYS, "n": len(out), "candles": len(df)}, "results": out}, fh, default=float) print(f"Guardado results.json ({len(out)} estrategias)") print("\nTOP 10 en train -> resultado en test (fuera de muestra):") for r in out[:10]: p, rt, rv = r["params"], r["train"], r["test"] print(f"{p}\n train: ret {rt['ret']:.1%} dd {rt['mdd']:.1%} trades {rt['trades']} liqs {rt['liqs']}" f"\n test : ret {rv['ret']:.1%} dd {rv['mdd']:.1%} trades {rv['trades']} liqs {rv['liqs']}\n") prof = sum(1 for r in out[:50] if r["test"]["ret"] > 0) print(f"De las 50 mejores en train, {prof} fueron positivas en test.")
